<script>
// راه‌اندازی SDK تلگرام
const tg = window.Telegram.WebApp;
tg.expand(); // تمام صفحه کردن مینی‌اپ
if (tg.initDataUnsafe && tg.initDataUnsafe.user) {
document.getElementById('player-name').innerText = tg.initDataUnsafe.user.first_name;
}

// وضعیت اولیه یا بازیابی از حافظه
let state = JSON.parse(localStorage.getItem('idleArenaSave')) || {
gold: 0,
stage: 1,
atkLvl: 1,
hpLvl: 1,
spdLvl: 1
};

let boostMultiplier = 1;
let boostTimeLeft = 0;

// محاسبه مشخصات هیرو بر اساس سطح ارتقا
function getHeroStats() {
return {
maxHp: 100 + (state.hpLvl - 1) * 30,
atk: (10 + (state.atkLvl - 1) * 5) * boostMultiplier,
attackInterval: Math.max(250, 1000 - (state.spdLvl - 1) * 60) // میلی‌ثانیه
};
}

// تولید دشمن متناسب با مرحله (Scaling)
const enemyIcons = ['👹', '👺', '🧟', '🧛', '🐉', '👾'];
function generateEnemy(stage) {
const isBoss = stage % 5 === 0;
const scale = Math.pow(1.18, stage - 1);
return {
name: isBoss ? `🔥 غول مرحله ${stage}` : `هیولای سطح ${stage}`,
icon: isBoss ? '👑' : enemyIcons[(stage - 1) % enemyIcons.length],
maxHp: Math.floor((isBoss ? 120 : 45) * scale),
currentHp: Math.floor((isBoss ? 120 : 45) * scale),
atk: Math.floor((isBoss ? 8 : 3) * Math.pow(1.15, stage - 1)),
reward: Math.floor((isBoss ? 35 : 10) * Math.pow(1.16, stage - 1))
};
}

let heroCurrentHp = getHeroStats().maxHp;
let currentEnemy = generateEnemy(state.stage);

// محاسبه هزینه ارتقاها
function getUpgradeCost(type) {
if (type === 'atk') return Math.floor(15 * Math.pow(1.25, state.atkLvl - 1));
if (type === 'hp') return Math.floor(15 * Math.pow(1.25, state.hpLvl - 1));
if (type === 'spd') return Math.floor(25 * Math.pow(1.35, state.spdLvl - 1));
}

function upgradeStat(type) {
const cost = getUpgradeCost(type);
if (state.gold >= cost) {
state.gold -= cost;
if (type === 'atk') state.atkLvl++;
if (type === 'hp') {
state.hpLvl++;
heroCurrentHp = getHeroStats().maxHp;
}
if (type === 'spd') {
state.spdLvl++;
restartCombatLoop();
}
if (tg.HapticFeedback) tg.HapticFeedback.impactOccurred('light'); // ویبره در تلگرام
saveGame();
updateUI();
}
}

function spawnDamageText(amount, isHeroTarget) {
const arena = document.getElementById('arena');
const el = document.createElement('div');
el.className = 'floating-dmg';
el.innerText = `-${amount}`;
el.style.left = isHeroTarget ? '25%' : '70%';
el.style.top = '35%';
if (isHeroTarget) el.style.color = '#ef4444';
arena.appendChild(el);
setTimeout(() => el.remove(), 700);
}

// حمله هیرو به دشمن
function heroAttack() {
const hero = getHeroStats();
currentEnemy.currentHp -= hero.atk;

const avatar = document.getElementById('hero-avatar');
avatar.classList.add('attack-anim');
setTimeout(() => avatar.classList.remove('attack-anim'), 100);
spawnDamageText(hero.atk, false);

if (currentEnemy.currentHp <= 0) {
// پیروزی در مرحله
state.gold += currentEnemy.reward * boostMultiplier;
state.stage++;
heroCurrentHp = Math.min(hero.maxHp, heroCurrentHp + Math.floor(hero.maxHp * 0.25)); // بازیابی ۲۵٪ جان
currentEnemy = generateEnemy(state.stage);
saveGame();
}
updateUI();
}

// حمله دشمن به هیرو (هر 1.2 ثانیه)
setInterval(() => {
heroCurrentHp -= currentEnemy.atk;
const avatar = document.getElementById('enemy-avatar');
avatar.classList.add('attack-anim');
setTimeout(() => avatar.classList.remove('attack-anim'), 100);
spawnDamageText(currentEnemy.atk, true);

if (heroCurrentHp <= 0) {
// شکست و بازگشت یک مرحله عقب‌تر (مکانیزم استاندارد Idle RPG)
state.stage = Math.max(1, state.stage - 1);
heroCurrentHp = getHeroStats().maxHp;
currentEnemy = generateEnemy(state.stage);
saveGame();
}
updateUI();
}, 1200);

// کلیک دستی روی میدان نبرد برای کمک به هیرو
document.getElementById('arena').addEventListener('click', () => {
heroAttack();
});

// حلقه حمله خودکار هیرو
let heroLoop;
function restartCombatLoop() {
clearInterval(heroLoop);
heroLoop = setInterval(heroAttack, getHeroStats().attackInterval);
}

// سیستم بوستر تبلیغاتی (اینجا کد SDK شبکه تبلیغاتی مثل Adsgram قرار می‌گیرد)
function activateAdBoost() {
if (boostTimeLeft > 0) return;
// شبیه‌سازی تماشای تبلیغ و دادن جایزه
boostMultiplier = 2;
boostTimeLeft = 60;
if (tg.HapticFeedback) tg.HapticFeedback.notificationOccurred('success');
updateUI();
}

setInterval(() => {
if (boostTimeLeft > 0) {
boostTimeLeft--;
const btn = document.getElementById('boost-btn');
btn.disabled = true;
btn.innerText = `⚡ بوست ۲ برابری فعال است (${boostTimeLeft} ثانیه)`;
if (boostTimeLeft === 0) {
boostMultiplier = 1;
btn.disabled = false;
btn.innerText = `🎬 تماشای تبلیغ: ۲ برابر قدرت و طلا (۶۰ ثانیه)`;
}
}
}, 1000);

function saveGame() {
localStorage.setItem('idleArenaSave', JSON.stringify(state));
}

function updateUI() {
const hero = getHeroStats();
document.getElementById('stage-num').innerText = state.stage;
document.getElementById('gold-num').innerText = state.gold.toLocaleString();

// هیرو
document.getElementById('hero-hp-bar').style.width = `${Math.max(0, (heroCurrentHp / hero.maxHp) * 100)}%`;
document.getElementById('hero-hp-text').innerText = `${Math.max(0, heroCurrentHp)}/${hero.maxHp}`;
document.getElementById('hero-atk-text').innerText = hero.atk;

// دشمن
document.getElementById('enemy-name').innerText = currentEnemy.name;
document.getElementById('enemy-avatar').innerText = currentEnemy.icon;
document.getElementById('enemy-hp-bar').style.width = `${Math.max(0, (currentEnemy.currentHp / currentEnemy.maxHp) * 100)}%`;
document.getElementById('enemy-hp-text').innerText = `${Math.max(0, currentEnemy.currentHp)}/${currentEnemy.maxHp}`;
document.getElementById('enemy-atk-text').innerText = currentEnemy.atk;

// ارتقاها
document.getElementById('atk-lvl').innerText = state.atkLvl;
document.getElementById('hp-lvl').innerText = state.hpLvl;
document.getElementById('spd-lvl').innerText = state.spdLvl;

const atkCost = getUpgradeCost('atk');
const hpCost = getUpgradeCost('hp');
const spdCost = getUpgradeCost('spd');

const btnAtk = document.getElementById('btn-upg-atk');
btnAtk.innerText = `${atkCost} 💰`;
btnAtk.disabled = state.gold < atkCost;

const btnHp = document.getElementById('btn-upg-hp');
btnHp.innerText = `${hpCost} 💰`;
btnHp.disabled = state.gold < hpCost;

const btnSpd = document.getElementById('btn-upg-spd');
btnSpd.innerText = `${spdCost} 💰`;
btnSpd.disabled = state.gold < spdCost;
}

// شروع بازی
restartCombatLoop();
updateUI();
</script>
